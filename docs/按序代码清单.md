## 3. 匹配规则引擎核心服务类

### backend/src/main/java/com/langchao/archive/engine/MatchingEngine.java

```java
package com.langchao.archive.engine;

import com.langchao.archive.model.MatchRule;
import com.langchao.archive.model.SourceDocument;
import com.langchao.archive.model.Voucher;
import org.springframework.stereotype.Component;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.LocalDate;
import java.time.temporal.ChronoUnit;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.HashSet;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Set;
import java.util.function.Function;

/**
 * 可配置规则引擎：策略负责候选计算；责任链按优先级消费资源。
 * 此类只计算建议，不替代数据库事务、行锁和 Redis 分布式锁。
 */
@Component
public class MatchingEngine {
    public record Proposal(List<Long> voucherIds, List<Long> documentIds, String ruleId,
                           BigDecimal confidenceScore, Map<String, Object> explanation) { }

    private interface Strategy {
        List<Proposal> propose(Context context, MatchRule rule, Options options);
    }

    private final Map<String, Strategy> strategies = Map.of(
        "R01", new ExactStrategy(Voucher::getInvoiceNo, SourceDocument::getInvoiceNo),
        "R02", new ExactStrategy(Voucher::getContractNo, SourceDocument::getContractNo),
        "R03", new ExactStrategy(Voucher::getBankFlowNo, SourceDocument::getBankFlowNo),
        "R04", new ExactStrategy(Voucher::getApprovalNo, SourceDocument::getApprovalNo),
        "R05", new CompositeStrategy(true),
        "R06", new CompositeStrategy(false),
        "R07", new SummaryStrategy());

    public List<Proposal> match(List<Voucher> vouchers, List<SourceDocument> documents, List<MatchRule> rules) {
        Objects.requireNonNull(vouchers, "vouchers");
        Objects.requireNonNull(documents, "documents");
        Objects.requireNonNull(rules, "rules");
        Context context = new Context(vouchers, documents);
        List<MatchRule> ordered = rules.stream().filter(rule -> Boolean.TRUE.equals(rule.getEnabled()))
            .sorted(Comparator.comparing(MatchRule::getPriority, Comparator.nullsLast(Integer::compareTo))
                .thenComparing(MatchRule::getRuleId)).toList();
        RuleNode head = null;
        // 反向组链；运行时 R01..R07 的实际先后由经过审核的 priority 控制。
        for (int index = ordered.size() - 1; index >= 0; index--) {
            MatchRule rule = ordered.get(index);
            Strategy strategy = strategies.get(rule.getRuleId());
            if (strategy == null) {
                throw new IllegalArgumentException("不支持的规则: " + rule.getRuleId());
            }
            head = new RuleNode(rule, strategy, head);
        }
        if (head != null) {
            head.execute(context);
        }
        return List.copyOf(context.accepted);
    }

    private record RuleNode(MatchRule rule, Strategy strategy, RuleNode next) {
        private void execute(Context context) {
            Options options = new Options(rule);
            List<Proposal> raw = strategy.propose(context, rule, options).stream()
                .filter(proposal -> proposal.confidenceScore().compareTo(options.threshold) >= 0).toList();
            // 同轮多个凭证争用同一单据时，所有冲突候选留在异常池，禁止按查询顺序“先到先得”。
            Map<Long, Integer> voucherUses = new HashMap<>();
            Map<Long, Integer> documentUses = new HashMap<>();
            for (Proposal proposal : raw) {
                proposal.voucherIds().forEach(id -> voucherUses.merge(id, 1, Integer::sum));
                proposal.documentIds().forEach(id -> documentUses.merge(id, 1, Integer::sum));
            }
            for (Proposal proposal : raw) {
                boolean conflict = proposal.voucherIds().stream().anyMatch(id -> voucherUses.get(id) > 1)
                    || proposal.documentIds().stream().anyMatch(id -> documentUses.get(id) > 1);
                if (conflict) {
                    context.blockedVouchers.addAll(proposal.voucherIds());
                    context.blockedDocuments.addAll(proposal.documentIds());
                }
            }
            for (Proposal proposal : raw) {
                if (proposal.voucherIds().stream().noneMatch(context.blockedVouchers::contains)
                    && proposal.documentIds().stream().noneMatch(context.blockedDocuments::contains)) {
                    context.accepted.add(proposal);
                    context.usedVouchers.addAll(proposal.voucherIds());
                    context.usedDocuments.addAll(proposal.documentIds());
                }
            }
            if (next != null) {
                next.execute(context);
            }
        }
    }

    private static final class Context {
        private final List<Voucher> vouchers;
        private final List<SourceDocument> documents;
        private final Set<Long> usedVouchers = new HashSet<>();
        private final Set<Long> usedDocuments = new HashSet<>();
        private final Set<Long> blockedVouchers = new HashSet<>();
        private final Set<Long> blockedDocuments = new HashSet<>();
        private final List<Proposal> accepted = new ArrayList<>();

        private Context(List<Voucher> vouchers, List<SourceDocument> documents) {
            if (vouchers.stream().anyMatch(value -> value.getId() == null)
                || vouchers.stream().map(Voucher::getId).distinct().count() != vouchers.size()
                || documents.stream().anyMatch(value -> value.getId() == null)
                || documents.stream().map(SourceDocument::getId).distinct().count() != documents.size()) {
                throw new IllegalArgumentException("匹配输入必须有唯一且非空的数据库主键");
            }
            this.vouchers = List.copyOf(vouchers);
            this.documents = List.copyOf(documents);
        }

        private List<Voucher> vouchers() {
            return vouchers.stream().filter(voucher -> pending(voucher.getStatus()))
                .filter(voucher -> !usedVouchers.contains(voucher.getId()) && !blockedVouchers.contains(voucher.getId())).toList();
        }

        private List<SourceDocument> documents() {
            return documents.stream().filter(document -> pending(document.getStatus()))
                .filter(document -> !usedDocuments.contains(document.getId()) && !blockedDocuments.contains(document.getId())).toList();
        }

        private static boolean pending(String status) {
            return status == null || "PENDING".equals(status);
        }

        private void ambiguous(Voucher voucher, List<SourceDocument> candidates) {
            blockedVouchers.add(voucher.getId());
            candidates.forEach(document -> blockedDocuments.add(document.getId()));
        }
    }

    private static final class Options {
        private final BigDecimal absolute;
        private final BigDecimal relative;
        private final double nameThreshold;
        private final int dateWindowDays;
        private final int maxCandidates;
        private final int maxNodes;
        private final BigDecimal threshold;

        private Options(MatchRule rule) {
            Map<String, Object> values = rule.getConditions() == null ? Map.of() : rule.getConditions();
            absolute = decimal(values, "absTolerance", "0.05");
            relative = decimal(values, "relativeTolerance", "0");
            nameThreshold = decimal(values, "nameThreshold", "0.85").doubleValue();
            dateWindowDays = decimal(values, "dateWindowDays", "7").intValueExact();
            // 配置必须实际约束搜索成本。预算用尽时不能据此宣称已找到唯一组合。
            maxCandidates = decimal(values, "maxCandidates", "20").intValueExact();
            maxNodes = decimal(values, "maxNodes", "100000").intValueExact();
            threshold = rule.getThreshold() == null ? new BigDecimal("0.85") : rule.getThreshold();
            if (absolute.signum() < 0 || relative.signum() < 0 || relative.compareTo(BigDecimal.ONE) > 0
                || nameThreshold < 0 || nameThreshold > 1 || dateWindowDays < 0 || dateWindowDays > 366
                || maxCandidates < 2 || maxCandidates > 20 || maxNodes < 1 || maxNodes > 100_000
                || threshold.signum() < 0 || threshold.compareTo(BigDecimal.ONE) > 0) {
                throw new IllegalArgumentException("规则阈值、日期窗口或金额容差超出允许范围");
            }
        }

        private static BigDecimal decimal(Map<String, Object> values, String key, String fallback) {
            Object value = values.get(key);
            return new BigDecimal(value == null ? fallback : value.toString());
        }

        private boolean amount(BigDecimal expected, BigDecimal actual) {
            return MoneyTolerance.within(expected, actual, absolute, relative);
        }

        private boolean date(LocalDate first, LocalDate second) {
            return first != null && second != null && Math.abs(ChronoUnit.DAYS.between(first, second)) <= dateWindowDays;
        }
    }

    private record ExactStrategy(Function<Voucher, String> voucherKey,
                                 Function<SourceDocument, String> documentKey) implements Strategy {
        @Override
        public List<Proposal> propose(Context context, MatchRule rule, Options options) {
            List<Proposal> result = new ArrayList<>();
            for (Voucher voucher : context.vouchers()) {
                String key = voucherKey.apply(voucher);
                if (key == null || key.isBlank()) {
                    continue;
                }
                List<SourceDocument> candidates = context.documents().stream()
                    .filter(document -> samePeriod(voucher, document))
                    // 精确标识符严格 String.equals；保留前导零，不做数字化或模糊比较。
                    .filter(document -> key.equals(documentKey.apply(document)))
                    // 标识符相同仍必须核对金额，不能因 OCR 号串碰撞而绕过金额校验。
                    .filter(document -> options.amount(voucher.getAmount(), document.getAmount())).toList();
                if (candidates.size() == 1) {
                    SourceDocument document = candidates.get(0);
                    result.add(proposal(List.of(voucher.getId()), List.of(document.getId()), rule,
                        new BigDecimal("0.99"), voucher.getAmount(), document.getAmount(), "EXACT_IDENTIFIER"));
                } else if (candidates.size() > 1) {
                    context.ambiguous(voucher, candidates);
                }
            }
            return result;
        }
    }

    private record CompositeStrategy(boolean exactName) implements Strategy {
        @Override
        public List<Proposal> propose(Context context, MatchRule rule, Options options) {
            List<Proposal> result = new ArrayList<>();
            for (Voucher voucher : context.vouchers()) {
                List<SourceDocument> candidates = context.documents().stream()
                    .filter(document -> samePeriod(voucher, document))
                    .filter(document -> options.amount(voucher.getAmount(), document.getAmount()))
                    .filter(document -> options.date(voucher.getVoucherDate(), document.getDocumentDate()))
                    .filter(document -> {
                        double nameScore = nameScore(voucher, document);
                        return exactName ? nameScore == 1 : nameScore >= options.nameThreshold;
                    }).toList();
                if (candidates.size() == 1) {
                    SourceDocument document = candidates.get(0);
                    // 金额、名称、日期先过硬门槛再计分；允许范围内的尾差与日期偏移降低置信度。
                    BigDecimal allowed = options.absolute.max(voucher.getAmount().abs().multiply(options.relative));
                    double amountQuality = allowed.signum() == 0 ? 1 : 1 - 0.1 * voucher.getAmount()
                        .subtract(document.getAmount()).abs().divide(allowed, 8, RoundingMode.HALF_UP).doubleValue();
                    long dayDelta = Math.abs(ChronoUnit.DAYS.between(voucher.getVoucherDate(), document.getDocumentDate()));
                    double dateQuality = 1 - 0.1 * dayDelta / (options.dateWindowDays + 1.0);
                    BigDecimal score = BigDecimal.valueOf(0.45 * amountQuality + 0.35 * nameScore(voucher, document) + 0.2 * dateQuality)
                        .setScale(4, RoundingMode.HALF_UP);
                    result.add(proposal(List.of(voucher.getId()), List.of(document.getId()), rule, score,
                        voucher.getAmount(), document.getAmount(), exactName ? "AMOUNT_NAME_DATE" : "FUZZY_NAME_DATE"));
                } else if (candidates.size() > 1) {
                    context.ambiguous(voucher, candidates);
                }
            }
            return result;
        }
    }

    private static final class SummaryStrategy implements Strategy {
        @Override
        public List<Proposal> propose(Context context, MatchRule rule, Options options) {
            List<Proposal> result = new ArrayList<>();
            for (Voucher voucher : context.vouchers()) {
                List<SourceDocument> candidates = context.documents().stream()
                    .filter(document -> eligible(voucher, document, options))
                    .filter(document -> document.getAmount() != null && document.getAmount().signum() > 0).toList();
                List<Long> ids = subset(candidates, SourceDocument::getId, SourceDocument::getAmount, voucher.getAmount(), options);
                if (!ids.isEmpty()) {
                    BigDecimal sum = candidates.stream().filter(document -> ids.contains(document.getId()))
                        .map(SourceDocument::getAmount).reduce(BigDecimal.ZERO, BigDecimal::add);
                    result.add(proposal(List.of(voucher.getId()), ids, rule, new BigDecimal("0.96"),
                        voucher.getAmount(), sum, "ONE_VOUCHER_MANY_DOCUMENTS"));
                }
            }
            for (SourceDocument document : context.documents()) {
                List<Voucher> candidates = context.vouchers().stream()
                    .filter(voucher -> eligible(voucher, document, options))
                    .filter(voucher -> voucher.getAmount() != null && voucher.getAmount().signum() > 0).toList();
                List<Long> ids = subset(candidates, Voucher::getId, Voucher::getAmount, document.getAmount(), options);
                if (!ids.isEmpty()) {
                    BigDecimal sum = candidates.stream().filter(voucher -> ids.contains(voucher.getId()))
                        .map(Voucher::getAmount).reduce(BigDecimal.ZERO, BigDecimal::add);
                    result.add(proposal(ids, List.of(document.getId()), rule, new BigDecimal("0.96"),
                        sum, document.getAmount(), "MANY_VOUCHERS_ONE_DOCUMENT"));
                }
            }
            return result;
        }

        private static boolean eligible(Voucher voucher, SourceDocument document, Options options) {
            return samePeriod(voucher, document) && options.date(voucher.getVoucherDate(), document.getDocumentDate())
                && nameScore(voucher, document) >= options.nameThreshold;
        }

        private static <T> List<Long> subset(List<T> items, Function<T, Long> id,
                                             Function<T, BigDecimal> amount, BigDecimal target, Options options) {
            if (target == null || target.signum() <= 0 || items.size() < 2 || items.size() > options.maxCandidates) {
                return List.of();
            }
            SubsetSearch<T> search = new SubsetSearch<>(items, id, amount, target, options);
            search.visit(0, BigDecimal.ZERO, new ArrayList<>());
            // 预算耗尽不能宣称“唯一解”；存在两个可行组合时同样转人工，避免任意选一组。
            return !search.exhausted && search.solutions.size() == 1 ? search.solutions.get(0) : List.of();
        }

        private static final class SubsetSearch<T> {
            private final List<T> items;
            private final Function<T, Long> id;
            private final Function<T, BigDecimal> amount;
            private final BigDecimal target;
            private final Options options;
            private final List<List<Long>> solutions = new ArrayList<>();
            private int nodes;
            private boolean exhausted;

            private SubsetSearch(List<T> items, Function<T, Long> id, Function<T, BigDecimal> amount,
                                 BigDecimal target, Options options) {
                this.items = items;
                this.id = id;
                this.amount = amount;
                this.target = target;
                this.options = options;
            }

            private void visit(int index, BigDecimal sum, List<Long> selected) {
                if (++nodes > options.maxNodes) {
                    exhausted = true;
                    return;
                }
                if (solutions.size() >= 2 || exhausted) {
                    return;
                }
                if (index == items.size()) {
                    if (selected.size() >= 2 && options.amount(target, sum)) {
                        solutions.add(List.copyOf(selected));
                    }
                    return;
                }
                BigDecimal allowed = options.absolute.max(target.abs().multiply(options.relative));
                if (sum.compareTo(target.add(allowed)) > 0) {
                    return;
                }
                T item = items.get(index);
                selected.add(id.apply(item));
                visit(index + 1, sum.add(amount.apply(item)), selected);
                selected.remove(selected.size() - 1);
                visit(index + 1, sum, selected);
            }
        }
    }

    private static boolean samePeriod(Voucher voucher, SourceDocument document) {
        return voucher.getPeriod() != null && !voucher.getPeriod().isBlank()
            && voucher.getPeriod().equals(document.getPeriod());
    }

    private static double nameScore(Voucher voucher, SourceDocument document) {
        String first = voucher.getNormalizedSupplierName();
        String second = document.getNormalizedSupplierName();
        return NameSimilarity.similarity(first == null || first.isBlank() ? voucher.getSupplierName() : first,
            second == null || second.isBlank() ? document.getSupplierName() : second);
    }

    private static Proposal proposal(List<Long> vouchers, List<Long> documents, MatchRule rule,
                                     BigDecimal confidence, BigDecimal expected, BigDecimal actual, String kind) {
        Map<String, Object> explanation = new LinkedHashMap<>();
        explanation.put("algorithm", kind);
        explanation.put("expectedAmount", expected.toPlainString());
        explanation.put("actualAmount", actual.toPlainString());
        explanation.put("amountDelta", expected.subtract(actual).abs().toPlainString());
        explanation.put("ruleVersion", rule.getVersion() == null ? 1 : rule.getVersion());
        return new Proposal(List.copyOf(vouchers), List.copyOf(documents), rule.getRuleId(), confidence, explanation);
    }
```

```java
package com.langchao.archive.engine;

import java.math.BigDecimal;

/** 使用 BigDecimal，禁止以 double 比较会计金额。 */
public final class MoneyTolerance {
    private MoneyTolerance() { }

    public static boolean within(BigDecimal expected, BigDecimal actual,
                                 BigDecimal absolute, BigDecimal relative) {
        if (expected == null || actual == null || absolute == null || relative == null) {
            return false;
        }
        if (absolute.signum() < 0 || relative.signum() < 0) {
            throw new IllegalArgumentException("金额容差不得为负");
        }
        // 两种容差任选满足一个即接受。相对容差以凭证期望金额绝对值为基数；零金额只能用绝对容差。
        BigDecimal allowed = absolute.max(expected.abs().multiply(relative));
        return expected.subtract(actual).abs().compareTo(allowed) <= 0;
    }
}
```

```java
package com.langchao.archive.engine;

import com.langchao.archive.util.OcrDataCleaner;

/** Levenshtein 编辑距离，滚动数组将空间复杂度降至 O(min(m,n))。 */
public final class NameSimilarity {
    private NameSimilarity() { }

    public static double similarity(String first, String second) {
        String left = OcrDataCleaner.normalizeName(first);
        String right = OcrDataCleaner.normalizeName(second);
        // 空值之间不能获得 100% 相似度，否则缺失客商的单据会被大规模错误匹配。
        if (left.isEmpty() || right.isEmpty()) {
            return 0;
        }
        if (left.equals(right)) {
            return 1;
        }
        int[] a = left.codePoints().toArray();
        int[] b = right.codePoints().toArray();
        if (a.length < b.length) {
            int[] temporary = a;
            a = b;
            b = temporary;
        }
        int[] previous = new int[b.length + 1];
        for (int index = 0; index <= b.length; index++) {
            previous[index] = index;
        }
        for (int row = 1; row <= a.length; row++) {
            int[] current = new int[b.length + 1];
            current[0] = row;
            for (int column = 1; column <= b.length; column++) {
                current[column] = Math.min(Math.min(current[column - 1] + 1, previous[column] + 1),
                    previous[column - 1] + (a[row - 1] == b[column - 1] ? 0 : 1));
            }
            previous = current;
        }
        return 1.0 - (double) previous[b.length] / Math.max(a.length, b.length);
    }
}
```

```java
import java.util.concurrent.atomic.AtomicBoolean;

/** Redis租约采用随机token + Lua校验续租/释放；数据库行锁仍是最终防重保障。 */
@Component
public class PeriodLock {
    private static final long LEASE_MS=30000;
    private final StringRedisTemplate redis;
    public PeriodLock(StringRedisTemplate redis) { this.redis=redis; }
    public <T> T execute(String period, Supplier<T> work) {
        String key="archive:lock:period:"+period, token=UUID.randomUUID().toString();
        if (!Boolean.TRUE.equals(redis.opsForValue().setIfAbsent(key,token,Duration.ofMillis(LEASE_MS)))) {
            throw new ApiException(409,"本期间正在执行匹配或归档，请稍后重试");
        }
        AtomicBoolean lost=new AtomicBoolean(false);
        ScheduledExecutorService watchdog=Executors.newSingleThreadScheduledExecutor(r->{Thread t=new Thread(r,"match-lease");t.setDaemon(true);return t;});
        ScheduledFuture<?> renewal=watchdog.scheduleAtFixedRate(()->{
            try {
                Long result=redis.execute(new DefaultRedisScript<>("if redis.call('get',KEYS[1])==ARGV[1] then return redis.call('pexpire',KEYS[1],ARGV[2]) else return 0 end",Long.class),List.of(key),token,String.valueOf(LEASE_MS));
                if(result==null||result==0) lost.set(true);
            } catch(RuntimeException e) { lost.set(true); }
        },10,10,TimeUnit.SECONDS);
        try {
            T result=work.get();
            // 即使租约丢失，业务也持有数据库行锁，不能出现双重分配。返回状态提示调用者刷新。
            if(lost.get()) throw new ApiException(409,"处理期间Redis租约中断，数据库已保持一致，请刷新结果");
            return result;
        } finally {
            renewal.cancel(false); watchdog.shutdownNow();
            try { redis.execute(new DefaultRedisScript<>("if redis.call('get',KEYS[1])==ARGV[1] then return redis.call('del',KEYS[1]) else return 0 end",Long.class),List.of(key),token); }
            catch(RuntimeException ignored) { /* 释放失败时由TTL自动回收，绝不删除其他持有者的锁。 */ }
        }
    }
}
```

```java
package com.langchao.archive.util;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.text.Normalizer;
import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.time.format.DateTimeParseException;
import java.time.format.ResolverStyle;
import java.util.Locale;
import java.util.regex.Pattern;

/** OCR 字段清洗。遇到歧义金额和无效日期拒绝入库，避免“清洗”产生另一笔金额。 */
public final class OcrDataCleaner {
    private static final Pattern MONEY = Pattern.compile("[+-]?(?:\\d{1,3}(?:,\\d{3})+|\\d+)(?:\\.\\d{1,2})?");
    private static final Pattern LEGAL_SUFFIX = Pattern.compile("(?:股份有限公司|集团有限公司|有限责任公司|股份公司|有限公司|集团|股份|公司)$");
    private static final DateTimeFormatter DATE = DateTimeFormatter.ofPattern("uuuu-M-d", Locale.ROOT)
        .withResolverStyle(ResolverStyle.STRICT);

    private OcrDataCleaner() { }

    public static BigDecimal amount(String raw) {
        if (raw == null || raw.isBlank()) {
            throw new IllegalArgumentException("金额不能为空");
        }
        String value = Normalizer.normalize(raw, Normalizer.Form.NFKC).trim()
            .replaceAll("(?i)(?:人民币|RMB|CNY|[¥￥元圆])", "").replaceAll("\\s+", "");
        boolean parenthesized = value.startsWith("(") && value.endsWith(")");
        if (parenthesized) {
            value = value.substring(1, value.length() - 1);
            if (value.startsWith("-") || value.startsWith("+")) {
                throw new IllegalArgumentException("金额的正负号有歧义");
            }
        }
        // 只接受一个完整数字；不可把“100.00 / 200.00”简单删除非数字字符后拼成新金额。
        if (!MONEY.matcher(value).matches()) {
            throw new IllegalArgumentException("金额格式无效或含有多个金额: " + raw);
        }
        BigDecimal result = new BigDecimal(value.replace(",", ""));
        if (result.precision() - result.scale() > 16) {
            throw new IllegalArgumentException("金额超出 DECIMAL(18,2) 范围");
        }
        return (parenthesized ? result.negate() : result).setScale(2, RoundingMode.UNNECESSARY);
    }

    public static LocalDate date(String raw) {
        if (raw == null || raw.isBlank()) {
            throw new IllegalArgumentException("日期不能为空");
        }
        String value = Normalizer.normalize(raw, Normalizer.Form.NFKC).trim()
            .replace("年", "-").replace("月", "-").replace("日", "")
            .replace('/', '-').replace('.', '-');
        try {
            if (value.matches("\\d{8}")) {
                return LocalDate.parse(value, DateTimeFormatter.BASIC_ISO_DATE);
            }
            return LocalDate.parse(value, DATE);
        } catch (DateTimeParseException exception) {
            throw new IllegalArgumentException("日期无效，要求真实的年月日: " + raw, exception);
        }
    }

    public static String normalizeName(String raw) {
        if (raw == null || raw.isBlank()) {
            return "";
        }
        String result = Normalizer.normalize(raw, Normalizer.Form.NFKC).toLowerCase(Locale.ROOT)
            .replaceAll("[\\s\\p{Punct}，。、《》【】·]+", "");
        // 仅移除末尾法律组织形式，不删除名称中间的“集团/股份”，防止不同主体误归一。
        String previous;
        do {
            previous = result;
            result = LEGAL_SUFFIX.matcher(result).replaceFirst("");
        } while (!previous.equals(result));
        return result;
    }
}
```

```java
    public ArchiveBook generate(String period, String startVoucherNo, String endVoucherNo, String actor) {
        VoucherNumber start = number(startVoucherNo), end = number(endVoucherNo);
        if (!start.prefix().equals(end.prefix()) || start.sequence() > end.sequence()) {
            throw new ApiException(400, "凭证号范围必须使用相同字别且起始序号不大于结束序号");
        }
        // 按固定顺序锁定所有凭证；与匹配写入共享数据库行锁。两个重叠档案任务不会重复归档。
        List<Voucher> selected = vouchers.selectList(new QueryWrapper<Voucher>().eq("period", period)
                .between("voucher_sequence", start.sequence(), end.sequence()).likeRight("voucher_no", start.prefix())
                .orderByAsc("id").last("FOR UPDATE"));
        selected = selected.stream().filter(v -> number(v.getVoucherNo()).prefix().equals(start.prefix()))
                .sorted(Comparator.comparing(Voucher::getVoucherSequence)).toList();
        if (selected.isEmpty() || selected.stream().noneMatch(v -> startVoucherNo.equals(v.getVoucherNo()))
                || selected.stream().noneMatch(v -> endVoucherNo.equals(v.getVoucherNo()))) {
            throw new ApiException(422, "所选期间中不存在完整的起止凭证");
        }
        if (selected.stream().anyMatch(v -> !"MATCHED".equals(v.getStatus()))) {
            throw new ApiException(409, "范围包含待匹配、附件缺失或已锁定凭证，无法归档");
        }
        List<Long> voucherIds = selected.stream().map(Voucher::getId).toList();
        List<MatchRelation> bindings = relations.selectList(new QueryWrapper<MatchRelation>()
                .in("voucher_id", voucherIds).orderByAsc("id"));
        Set<Long> documentIds = bindings.stream().map(MatchRelation::getDocId).collect(Collectors.toSet());
        if (documentIds.isEmpty()) { throw new ApiException(422, "凭证缺少原始单据"); }
        List<SourceDocument> lockedDocuments = documents.selectList(new QueryWrapper<SourceDocument>()
                .in("id", documentIds).orderByAsc("id").last("FOR UPDATE"));
        Map<Long, SourceDocument> byId = lockedDocuments.stream().collect(Collectors.toMap(SourceDocument::getId, Function.identity()));
        if (byId.size() != documentIds.size()) { throw new ApiException(422, "关联单据已不存在"); }
        Map<Long, List<MatchRelation>> byVoucher = bindings.stream().collect(Collectors.groupingBy(MatchRelation::getVoucherId));
        List<Path> inputs = new ArrayList<>();
        for (Voucher voucher : selected) {
            List<MatchRelation> attachments = byVoucher.getOrDefault(voucher.getId(), List.of());
            validateVoucher(voucher, attachments);
            // 每张记账凭证后紧接其原始单据，单据按类型、业务日期和主键稳定排序。
            inputs.add(requiredFile(voucher.getImageUrl(), "凭证 " + voucher.getVoucherNo()));
            List<SourceDocument> ordered = attachments.stream().map(r -> byId.get(r.getDocId())).distinct()
                    .sorted(Comparator.comparingInt((SourceDocument d) -> documentOrder(d.getDocType()))
                            .thenComparing(SourceDocument::getDocumentDate, Comparator.nullsLast(Comparator.naturalOrder()))
                            .thenComparing(SourceDocument::getId)).toList();
            for (SourceDocument document : ordered) {
                if ("INVOICE".equalsIgnoreCase(document.getDocType()) && !Boolean.TRUE.equals(document.getSealPresent())) {
                    throw new ApiException(422, "发票 " + document.getId() + " 缺少已核验的有效印章");
                }
                if (!Set.of("MATCHED", "LOCKED").contains(Objects.toString(document.getStatus(), ""))) {
                    throw new ApiException(422, "单据 " + document.getId() + " 尚未完成匹配");
                }
                Path image = requiredFile(document.getImageUrl(), "单据 " + document.getId());
                verifySourceHash(document, image);
                inputs.add(image);
            }
        }
        // 多凭证共用同一单据时，所有分摊额之和必须与单据面额一致，防止重复分配。
        for (SourceDocument document : lockedDocuments) {
            BigDecimal allocated = relations.selectList(new QueryWrapper<MatchRelation>().eq("doc_id", document.getId()))
                    .stream().map(MatchRelation::getAllocatedAmount).reduce(BigDecimal.ZERO, BigDecimal::add);
            if (document.getAmount() == null || allocated.subtract(document.getAmount()).abs().compareTo(AMOUNT_TOLERANCE) > 0) {
                throw new ApiException(422, "单据 " + document.getId() + " 的分摊金额与原始面额不一致");
            }
        }
        String outputUrl = null;
        Path temporary = null;
        try {
            temporary = storage.temporary();
            merge(inputs, temporary);
            String hash = sha256(temporary);
            outputUrl = storage.commit(temporary, "pdf");
            String publishedUrl = outputUrl;
            // 数据库提交失败时移除新档案，避免出现没有数据库记录的孤立文件。
            if (TransactionSynchronizationManager.isSynchronizationActive()) {
                TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
                    @Override public void afterCompletion(int status) {
                        if (status != STATUS_COMMITTED) { storage.delete(publishedUrl); }
                    }
                });
            }
            ArchiveBook book = new ArchiveBook();
            book.setPeriod(period); book.setStartVoucherNo(startVoucherNo); book.setEndVoucherNo(endVoucherNo);
            book.setStartVoucherSequence(start.sequence()); book.setEndVoucherSequence(end.sequence());
            book.setPdfUrl(outputUrl); book.setHashCode(hash); book.setStatus("LOCKED"); book.setCreatedAt(LocalDateTime.now());
            book.setVoucherCount(selected.size()); book.setDocumentCount(documentIds.size()); book.setCreator(actor);
            archives.insert(book);
            vouchers.update(null, new UpdateWrapper<Voucher>().in("id", voucherIds).set("status", "LOCKED"));
            documents.update(null, new UpdateWrapper<SourceDocument>().in("id", documentIds).set("status", "LOCKED"));
            SysLog log = new SysLog(); log.setAction("ARCHIVE_GENERATE"); log.setTargetType("ARCHIVE_BOOK");
            log.setTargetId(book.getId().toString()); log.setOperator(actor); log.setDetail("SHA-256=" + hash);
            log.setCreatedAt(LocalDateTime.now()); logs.insert(log);
            return book;
        } catch (IOException exception) {
            if (outputUrl != null) { storage.delete(outputUrl); }
            throw new ApiException(422, "档案影像不可读取或 PDF 合并失败");
        } catch (RuntimeException exception) {
            if (outputUrl != null && !TransactionSynchronizationManager.isSynchronizationActive()) { storage.delete(outputUrl); }
            throw exception;
        } finally {
            if (temporary != null) {
                try { Files.deleteIfExists(temporary); } catch (IOException ignored) { /* 已发布文件另有事务回滚清理。 */ }
            }
        }
    }

            where.append(" AND EXISTS (SELECT 1 FROM voucher v WHERE v.period=a.period AND v.voucher_sequence BETWEEN a.start_voucher_sequence AND a.end_voucher_sequence");

OR d.supplier_name LIKE ?)))");

    public Resource pdf(long id) {

                throw new ApiException(409, "档案完整性校验失败，文件可能已被修改");

    private void validateVoucher(Voucher voucher, List<MatchRelation> attachments) {
        long actual = attachments.stream().map(MatchRelation::getDocId).distinct().count();
        if (actual == 0 || voucher.getExpectedAttachmentCount() == null || actual != voucher.getExpectedAttachmentCount()) {
            throw new ApiException(422, "凭证 " + voucher.getVoucherNo() + " 附件数量与预期不一致");
        }
        if (attachments.stream().anyMatch(r -> r.getAllocatedAmount() == null || r.getAllocatedAmount().signum() < 0)) {
            throw new ApiException(422, "凭证存在无效分摊金额");
        }
        BigDecimal amount = attachments.stream().map(MatchRelation::getAllocatedAmount).reduce(BigDecimal.ZERO, BigDecimal::add);
        // BigDecimal 做十进制相减，禁止 double 引入 0.01 尾差附近的二进制舍入误判。
        if (voucher.getAmount() == null || amount.subtract(voucher.getAmount()).abs().compareTo(AMOUNT_TOLERANCE) > 0) {
            throw new ApiException(422, "凭证 " + voucher.getVoucherNo() + " 附件金额不一致（允许尾差 0.05 元）");
        }
    }

    private void verifySourceHash(SourceDocument document, Path file) {
        if (!hasText(document.getContentHash())) { return; }
        try {
            if (!MessageDigest.isEqual(sha256(file).getBytes(java.nio.charset.StandardCharsets.US_ASCII),
                    document.getContentHash().getBytes(java.nio.charset.StandardCharsets.US_ASCII))) {
                throw new ApiException(409, "单据 " + document.getId() + " 影像哈希与采集记录不一致，禁止归档");
            }
        } catch (IOException exception) { throw new ApiException(422, "无法核验原始单据影像"); }
    }

    private Path requiredFile(String url, String label) {
        if (url == null || url.isBlank()) { throw new ApiException(422, label + " 缺少影像，不能生成档案"); }
        Path file = storage.resolve(url);
        if (!Files.isRegularFile(file) || !Files.isReadable(file)) { throw new ApiException(422, label + " 影像文件不存在"); }
        return file;
    }

    /** PDF 原页无损导入；PNG/JPEG 按比例居中放入 A4。最终顺序与 inputs 完全一致。 */
    public static void merge(List<Path> inputs, Path output) throws IOException {
        if (inputs.isEmpty()) { throw new ApiException(422, "没有可合并的影像"); }
        try (PDDocument result = new PDDocument()) {
            for (Path input : inputs) {
                if (input.getFileName().toString().toLowerCase(java.util.Locale.ROOT).endsWith(".pdf")) {
                    try (PDDocument source = Loader.loadPDF(input.toFile())) {
                        if (source.isEncrypted() || source.getNumberOfPages() == 0) { throw new ApiException(422, "加密或空白 PDF 不能直接归档"); }
                        if (result.getNumberOfPages() + source.getNumberOfPages() > MAX_PAGES) {
                            throw new ApiException(422, "单册超过 5000 页，请缩小凭证范围");
                        }
                        // appendDocument 深拷贝资源，源 PDF 关闭后字体/图片流仍然可用。
                        new PDFMergerUtility().appendDocument(result, source);
                    }
                } else {
                    PDImageXObject image = PDImageXObject.createFromFileByContent(input.toFile(), result);
                    if (image.getWidth() <= 0 || image.getHeight() <= 0 || (long) image.getWidth() * image.getHeight() > 80_000_000L) {
                        throw new ApiException(422, "影像尺寸无效或超过 8000 万像素");
                    }
                    if (result.getNumberOfPages() >= MAX_PAGES) { throw new ApiException(422, "单册超过 5000 页"); }
                    PDPage page = new PDPage(PDRectangle.A4); result.addPage(page);
                    float margin = 24, width = page.getMediaBox().getWidth() - 2 * margin, height = page.getMediaBox().getHeight() - 2 * margin;
                    float scale = Math.min(width / image.getWidth(), height / image.getHeight());
                    float renderedWidth = image.getWidth() * scale, renderedHeight = image.getHeight() * scale;
                    try (PDPageContentStream stream = new PDPageContentStream(result, page)) {
                        stream.drawImage(image, (page.getMediaBox().getWidth() - renderedWidth) / 2,
                                (page.getMediaBox().getHeight() - renderedHeight) / 2, renderedWidth, renderedHeight);
                    }
                }
            }
            result.save(output.toFile());
        }
    }

    public static String sha256(Path file) throws IOException {
        try {
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            try (InputStream input = new DigestInputStream(Files.newInputStream(file), digest)) { input.transferTo(java.io.OutputStream.nullOutputStream()); }
            return HexFormat.of().formatHex(digest.digest());
        } catch (NoSuchAlgorithmException exception) { throw new IllegalStateException("JVM 不支持 SHA-256", exception); }
    }

    private int documentOrder(String type) {
        return switch (Objects.toString(type, "")) { case "INVOICE" -> 1; case "BANK_RECEIPT" -> 2; case "CONTRACT" -> 3; case "APPROVAL" -> 4; default -> 5; };
    }

    private VoucherNumber number(String value) {
        Matcher matcher = NUMBER.matcher(Objects.toString(value, ""));
        if (!matcher.matches()) { throw new ApiException(400, "凭证号必须以数字序号结尾"); }
        try { return new VoucherNumber(matcher.group(1), Integer.parseInt(matcher.group(2))); }
        catch (NumberFormatException exception) { throw new ApiException(400, "凭证序号超出范围"); }
    }
```

```java
@Component
public class FileStorage {
    private static final long MAX_BYTES = 20L * 1024 * 1024;
    private static final Pattern INTERNAL_PATH = Pattern.compile("^/files/[0-9a-fA-F-]{36}\\.(pdf|png|jpg)$");
    private final Path root;

    public FileStorage(@Value("${archive.storage-root:./data}") String directory) {
        try {
            Path configured = Path.of(directory).toAbsolutePath().normalize();
            Files.createDirectories(configured);
            this.root = configured.toRealPath();
        } catch (IOException exception) {
            throw new IllegalStateException("无法初始化文件存储目录", exception);
        }
    }

    /** 文件名由服务器生成，并按文件头识别格式，不信任上传方文件名和 Content-Type。 */
    public String store(MultipartFile upload) {
        if (upload == null || upload.isEmpty() || upload.getSize() > MAX_BYTES) {
            throw new ApiException(400, "文件不能为空且必须小于 20MB");
        }
        Path temporary = null;
        try (InputStream input = upload.getInputStream()) {
            temporary = temporary();
            long copied;
            try (var output = Files.newOutputStream(temporary, StandardOpenOption.TRUNCATE_EXISTING)) {
                copied = input.transferTo(new java.io.FilterOutputStream(output) {
                    private long count;
                    @Override public void write(byte[] bytes, int offset, int length) throws IOException {
                        count += length;
                        if (count > MAX_BYTES) { throw new IOException("文件超过20MB限制"); }
                        out.write(bytes, offset, length);
                    }
                    @Override public void write(int value) throws IOException {
                        if (++count > MAX_BYTES) { throw new IOException("文件超过20MB限制"); }
                        out.write(value);
                    }
                });
            }
            if (copied == 0) { throw new ApiException(400, "文件为空"); }
            byte[] header;
            try (InputStream saved = Files.newInputStream(temporary)) { header = saved.readNBytes(8); }
            String extension = extension(header);
            return commit(temporary, extension);
        } catch (IOException exception) {
            throw new ApiException(400, "上传失败：文件不可读取或超过大小限制");
        } finally {
            if (temporary != null) {
                try { Files.deleteIfExists(temporary); } catch (IOException ignored) { /* 临时文件清理由运维兜底。 */ }
            }
        }
    }

    public Path temporary() throws IOException { return Files.createTempFile(root, "archive-", ".tmp"); }

    public String commit(Path temporary, String extension) throws IOException {
        if (!java.util.Set.of("pdf", "png", "jpg").contains(extension.toLowerCase(Locale.ROOT))) {
            throw new ApiException(400, "不支持的文件格式");
        }
        String url = "/files/" + UUID.randomUUID() + "." + extension;
        Path destination = resolve(url);
        // 临时文件与最终文件位于同一文件系统，原子发布避免读到未写完的档案册。
        try { Files.move(temporary, destination, StandardCopyOption.ATOMIC_MOVE); }
        catch (java.nio.file.AtomicMoveNotSupportedException exception) { Files.move(temporary, destination); }
        return url;
    }

    public Path resolve(String internalUrl) {
        if (internalUrl == null || !INTERNAL_PATH.matcher(internalUrl).matches()) {
            throw new ApiException(400, "文件路径必须是系统内部文件标识");
        }
        Path resolved = root.resolve(internalUrl.substring("/files/".length())).normalize();
        if (!resolved.startsWith(root) || Files.isSymbolicLink(resolved)) {
            throw new ApiException(400, "非法文件路径");
        }
        return resolved;
    }

    public Resource resource(String internalUrl) {
        Path file = resolve(internalUrl);
        if (!Files.isRegularFile(file) || !Files.isReadable(file)) { throw new ApiException(404, "影像文件不存在"); }
        try { return new UrlResource(file.toUri()); }
        catch (java.net.MalformedURLException exception) { throw new ApiException(500, "文件地址无效"); }
    }

        try { Files.deleteIfExists(resolve(internalUrl)); }
        catch (IOException exception) { throw new ApiException(500, "清理文件失败"); }
    }

    private String extension(byte[] header) {
        if (header.length >= 5 && header[0] == '%' && header[1] == 'P' && header[2] == 'D'
                && header[3] == 'F' && header[4] == '-') { return "pdf"; }
        if (header.length >= 8 && (header[0] & 255) == 137 && header[1] == 'P' && header[2] == 'N'
                && header[3] == 'G' && header[4] == 13 && header[5] == 10 && header[6] == 26 && header[7] == 10) { return "png"; }
        if (header.length >= 3 && (header[0] & 255) == 255 && (header[1] & 255) == 216 && (header[2] & 255) == 255) { return "jpg"; }
        throw new ApiException(400, "仅接受 PDF、PNG、JPEG 影像");
    }
```

```java
package com.langchao.archive.util;

import org.junit.jupiter.api.Test;
import org.junit.jupiter.params.ParameterizedTest;
import org.junit.jupiter.params.provider.CsvSource;
import org.springframework.data.redis.core.HashOperations;
import org.springframework.data.redis.core.StringRedisTemplate;

import java.math.BigDecimal;
import java.time.LocalDate;

import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

class OcrDataCleanerTest {
    @ParameterizedTest
    @CsvSource(delimiter = '|', value = {
        "¥1,234.56|1234.56", "人民币 88.01 元|88.01", "￥１２３．４５|123.45",
        "RMB 0.01|0.01", "(¥20.10)|-20.10", "-18.01|-18.01", "0|0.00"
    })
    void cleanCurrencyWithoutLosingCents(String input, String expected) {
        assertEquals(new BigDecimal(expected), OcrDataCleaner.amount(input));
    }

    @Test
    void rejectAmbiguousOrMalformedAmounts() {
        for (String input : new String[]{"100.00 / 200.00", "1,23.00", "1.001", "1e3", "(-20)", "", "abc"}) {
            assertThrows(IllegalArgumentException.class, () -> OcrDataCleaner.amount(input), input);
        }
        assertThrows(IllegalArgumentException.class, () -> OcrDataCleaner.amount(null));
    }

    @ParameterizedTest
    @CsvSource({"2026年10月2日,2026-10-02", "2026/10/02,2026-10-02", "20261002,2026-10-02", "2026.10.2,2026-10-02"})
    void normalizeDates(String input, String expected) {
        assertEquals(LocalDate.parse(expected), OcrDataCleaner.date(input));
    }

    @Test
    void rejectImpossibleDates() {
        assertThrows(IllegalArgumentException.class, () -> OcrDataCleaner.date("2026-02-29"));
        assertThrows(IllegalArgumentException.class, () -> OcrDataCleaner.date("2026-13-01"));
    }

    @Test
    void removeLegalSuffixOnly() {
        assertEquals("阿里云", OcrDataCleaner.normalizeName("阿里云有限公司"));
        assertEquals("华北", OcrDataCleaner.normalizeName(" 华北集团股份有限公司 "));
        assertEquals("集团研究院", OcrDataCleaner.normalizeName("集团研究院有限公司"));
    }

    @Test
    @SuppressWarnings("unchecked")
    void normalizeApprovedAliasFromRedis() {
        StringRedisTemplate redis = mock(StringRedisTemplate.class);
        HashOperations<String, Object, Object> hash = mock(HashOperations.class);
        when(redis.opsForHash()).thenReturn(hash);
        when(hash.get(SupplierNormalizer.ALIAS_KEY, "阿里云计算")).thenReturn("阿里云");
        assertEquals("阿里云", new SupplierNormalizer(redis).normalize("阿里云计算有限公司"));
        verify(hash).get(SupplierNormalizer.ALIAS_KEY, "阿里云计算");
    }
}
```
